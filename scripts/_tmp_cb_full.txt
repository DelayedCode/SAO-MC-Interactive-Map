 703:     const clampOffset = () => {
 704:       const minX = Math.min(0, viewport.clientWidth - 921 * skillTreeViewState.scale);
 705:       const minY = Math.min(0, viewport.clientHeight - 923 * skillTreeViewState.scale);

 713:       event.preventDefault();
 714:       const rect = viewport.getBoundingClientRect();

 716:       const pointerY = event.clientY - rect.top;
 717:       const previousScale = skillTreeViewState.scale;
 718:       const nextScale = Math.min(2, Math.max(0.4, previousScale * (event.deltaY < 0 ? 1.12 : 0.89)));
 719:       const worldX = (pointerX - skillTreeViewState.offsetX) / previousScale;
 720:       const worldY = (pointerY - skillTreeViewState.offsetY) / previousScale;
