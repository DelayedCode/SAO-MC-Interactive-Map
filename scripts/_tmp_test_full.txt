  55: function pruneExpectedSelection(selected, adjacency) {
  56:   const result = new Set(selected);
  57:   let removed;
  58:   do {
  59:     removed = false;
  60:     for (const skillId of result) {
  61:       if (skillId === "skill1") continue;
  62:       const valid = skillId === "skill58"
  63:         ? screenshotThreePrerequisites.every((id) => result.has(id))
  64:         : adjacency[skillId].some((id) => result.has(id));
  65:       if (!valid) {
  66:         result.delete(skillId);
  67:         removed = true;
  68:       }
  69:     }
  70:   } while (removed);
  71:   return result;
  72: }

  74: async function assertRenderedStates(page, classId, selected, adjacency) {
  75:   const actual = await page.locator("#skillTreeCanvas .skill-node").evaluateAll((nodes) =>
  76:     nodes.map((node) => ({
  77:       id: node.dataset.skillId.replace(`${node.dataset.skillId.split("-")[0]}-`, ""),
  78:       selected: node.classList.contains("is-selected"),
  79:       available: node.classList.contains("is-available"),
  80:       locked: node.classList.contains("is-locked")
  81:     }))
  82:   );
  83:   const selectedWithRoot = new Set(selected);
  84:   selectedWithRoot.add("skill1");
  85:   for (const node of actual) {
  86:     const isSelected = selectedWithRoot.has(node.id);
  87:     const isAvailable = node.id === "skill58"
  88:       ? screenshotThreePrerequisites.every((id) => selectedWithRoot.has(id))
  89:       : adjacency[node.id].some((id) => selectedWithRoot.has(id));
  90:     assert.equal(node.selected, isSelected, `${classId} ${node.id}: selected state matches the model`);
  91:     assert.equal(node.available, !isSelected && isAvailable, `${classId} ${node.id}: availability matches the model`);
  92:     assert.equal(node.locked, !isSelected && !isAvailable, `${classId} ${node.id}: lock state matches the model`);
  93:   }
  94: }
